### 7.3 Social Sharing Image Optimization

   - PNG compresses poorly for photos

   - Increases bandwidth usage

# Using cwebp (Google's tool)

#### 8.3.2 Tag Dynamic Route

```astro
// src/pages/notes/tags/[tag].astro
export async function getStaticPaths() {
  const notes = await getCollection('notes');
  const tagMap = new Map<string, typeof notes>();

  for (const note of notes) {
    for (const tag of note.data.tags ?? []) {
      if (!tagMap.has(tag)) tagMap.set(tag, []);
      tagMap.get(tag)!.push(note);
    }
  }

  return [...tagMap.entries()].map(([tag, notes]) => ({
    params: { tag },
    props: {
      tag,
      notes: notes.sort((a, b) => b.data.date.getTime() - a.data.date.getTime()),
    },
  }));
}
```
